cmake_minimum_required(VERSION 3.16)
project(hf_native_capture VERSION 0.1.0 LANGUAGES C CXX)

option(HF_BUILD_WORKER "Build the synthetic capture worker and DNG writer" ON)
option(HF_BUILD_DAEMON "Build the native synthetic camera authority" ON)
option(HF_BUILD_IMAGE_PROCESSOR "Build the bounded OpenCV 4.12 image processor" OFF)

find_package(Threads REQUIRED)
add_library(hf_capture src/core.cpp)
target_include_directories(hf_capture PUBLIC include)
target_compile_features(hf_capture PUBLIC cxx_std_17)
set_target_properties(hf_capture PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(hf_capture PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(hf_capture PRIVATE /W4 /permissive- /fp:strict)
else()
    target_compile_options(hf_capture PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()

if(HF_BUILD_IMAGE_PROCESSOR)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../processing" "${CMAKE_CURRENT_BINARY_DIR}/processing")
endif()

enable_testing()
add_executable(hf-core-tests tests/test_core.cpp)
target_link_libraries(hf-core-tests PRIVATE hf_capture)
add_test(NAME native-core COMMAND hf-core-tests)

if(HF_BUILD_WORKER)
    add_executable(hf-native-worker src/worker.cpp src/dng.cpp)
    target_link_libraries(hf-native-worker PRIVATE hf_capture)
    if(MSVC)
        target_compile_options(hf-native-worker PRIVATE /W4 /permissive-)
    else()
        target_compile_options(hf-native-worker PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(MINGW)
        target_link_options(hf-native-worker PRIVATE -municode)
    endif()
endif()

if(HF_BUILD_DAEMON)
    add_library(hf_sqlite STATIC third_party/sqlite/sqlite3.c)
    target_include_directories(hf_sqlite PUBLIC third_party/sqlite)
    target_compile_definitions(hf_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0)
    target_link_libraries(hf_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
    add_library(hf_daemon src/daemon.cpp src/dng.cpp src/sha256.cpp src/calibration.cpp)
    target_include_directories(hf_daemon PUBLIC include third_party)
    target_compile_features(hf_daemon PUBLIC cxx_std_17)
    set_target_properties(hf_daemon PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(hf_daemon PUBLIC hf_capture hf_sqlite)
    if(WIN32)
        target_link_libraries(hf_daemon PRIVATE bcrypt)
    endif()
    add_executable(hf-camerad src/camerad.cpp)
    target_link_libraries(hf-camerad PRIVATE hf_daemon)
    add_executable(hf-daemon-tests tests/test_daemon.cpp)
    target_link_libraries(hf-daemon-tests PRIVATE hf_daemon)
    add_test(NAME native-authority COMMAND hf-daemon-tests)
    add_executable(hf-sha256-tests tests/test_sha256.cpp)
    target_link_libraries(hf-sha256-tests PRIVATE hf_daemon)
    add_test(NAME native-sha256 COMMAND hf-sha256-tests)
    add_executable(hf-calibration-tests tests/test_calibration.cpp)
    target_link_libraries(hf-calibration-tests PRIVATE hf_daemon)
    add_test(NAME native-calibration COMMAND hf-calibration-tests "${CMAKE_CURRENT_SOURCE_DIR}/../calibration/fixtures")
    if(MINGW)
        target_link_options(hf-camerad PRIVATE -municode)
    endif()
    if(MSVC)
        target_compile_options(hf_daemon PRIVATE /W4 /permissive-)
        target_compile_options(hf-camerad PRIVATE /W4 /permissive-)
    else()
        target_compile_options(hf_daemon PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(hf-camerad PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
